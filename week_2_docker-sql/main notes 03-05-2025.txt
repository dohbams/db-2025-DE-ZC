-- this works to create a db

docker run -it \
    -e POSTGRES_USER="root" \
    -e POSTGRES_PASSWORD="root" \
    -e POSTGRES_DB="ny_taxi_db" \
    -v ny_taxi_data:/var/lib/postgresql/data \
    -p 5432:5432 \
    postgres:15


-- this is for connecting to the db

pgcli -h localhost -p 5432 -u root -d ny_taxi_db



-- this is for starting pgadmin 4 GUI

docker run -it \
    -e PGADMIN_DEFAULT_EMAIL="admin@admin.com" \
    -e PGADMIN_DEFAULT_PASSWORD="root" \
    -p 8080:80 \
    dpage/pgadmin4


-- this to create a docker network (docker network is used so that two containers can interact as long as
they are in the same network)

docker network create pg-network (pg-network is the name of the network)

-- here we recreate our db container inside the network we created

docker run -it \
    -e POSTGRES_USER="root" \
    -e POSTGRES_PASSWORD="root" \
    -e POSTGRES_DB="ny_taxi_db" \
    -v ny_taxi_data:/var/lib/postgresql/data \
    -p 5432:5432 \
    --network=pg-network \
    --name pg-database \
    postgres:13

-- here we now run pgadmin in the same network as wekll

    docker run -it \
    -e PGADMIN_DEFAULT_EMAIL="admin@admin.com" \
    -e PGADMIN_DEFAULT_PASSWORD="root" \
    -p 8080:80 \
    --network=pg-network \
    --name pgadmin \
    dpage/pgadmin4


-- here we construct the way our python script will be executed

-- this works to ingest data using the ingest_data.py script

URL="https://d37ci6vzurychx.cloudfront.net/trip-data/yellow_tripdata_2024-01.parquet"

python ingest_data.py \
    --user=root \
    --password=root \
    --host=localhost \
    --port=5432 \
    --db=ny_taxi_db \
    --table_name=yellow_taxi_data \
    --url="$URL"

-- we updated our dockerfile to make sure it runs our python script and we also installed some more dependencies like wget, sqlalchemy and psycopg2

-- now we build the container using the docker file inside our directory
docker build -t taxi_ingest:v001 

-- here we run the container and supply our parameters as set in ingest_data.py
-- we make sure to run it in our created network so that it can interact with the existing db also change host to the name of the db container

URL="https://d37ci6vzurychx.cloudfront.net/trip-data/yellow_tripdata_2024-01.parquet"

docker run -it \
    --network=pg-network \
    taxi_ingest:v001 \
        --user=root \
        --password=root \
        --host=pg-database \
        --port=5432 \
        --db=ny_taxi_db \
        --table_name=yellow_taxi_data \
        --url="$URL"



-- docker-compose.yaml

services:
  pgdatabase:
    image: postgres:13
    environment:
      - POSTGRES_USER=root
      - POSTGRES_PASSWORD=root
      - POSTGRES_DB=ny_taxi_db

    volumes:
      - "./ny_taxi_data:/var/lib/postgresql/data:rw"
    
    ports:
      - "5432:5432"

  pgadmin:
    image: dpage/pgadmin4

    environment:
      - PGADMIN_DEFAULT_EMAIL=admin@admin.com
      - PGADMIN_DEFAULT_PASSWORD=root

    ports:
      - "8080:80"
