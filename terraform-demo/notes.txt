fiest we wnt to GCP loggedd in and created a service account.
this service account basically helps us to be able to run several GCP services like bigquery, cloud storage etc.
created a terraform service account and added the needed roles based on the resources we want to manage using terraform.
we also created credential keys (which will probably use in terraform) to authenicate the service account for managing the resources.
this credential is saved as keys.json in the keys folder in our current directory (terraform-demo)


we use this in our main.tf file to set up the provider. we're using ggoogle cloud here

terraform {
  required_providers {
    google = {
      source  = "hashicorp/google"
      version = "6.26.0"
    }
  }
}

provider "google" {
  # Configuration options
  project = "bigquery-portfolio"
  region  = "europe-west1"
}

for our credentials we use this bash command export GOOGLE_CREDENTIALS="keys/keys.json" so that we set the keys.json as an environment variable 
to point to the Google Cloud service account key file. 

we then use echo $GOOGLE_CREDENTIALS to confirm that the environment variable is set correctly.

we use the command terraform init to initialize terraform

WE CAN ALSo use terraform with variables as this can be very helpful. we can set variables like region, location, bucket name etc

here's our initial main.tf script
terraform {
  required_providers {
    google = {
      source  = "hashicorp/google"
      version = "6.26.0"
    }
  }
}

provider "google" {
  project = "bigquery-portfolio"
  region  = "europe-west1"
}

resource "google_storage_bucket" "tf-demo-bucket" {
  name          = "bigquery-portfolio-tf-demo-bucket"
  location      = "EU"
  storage_class = "STANDARD"
  force_destroy = true
  public_access_prevention = "enforced"

  lifecycle_rule {
    condition {
      age = 1
    }
    action {
      type = "AbortIncompleteMultipartUpload"
    }
  }
}

here are our defined variables in variables.tf

variable "project" {
  description = "Project Name"
  default     = "bigquery-portfolio"
}

variable "region" {
  description = "Project Region"
  default     = "europe-west1"
}


variable "location" {
  description = "Project Location"
  default     = "EU"
}

variable "credentials" {
  description = "Credentials"
  default =  "./keys/keys.json"
  sensitive = true
}



variable "bq_dataset_name" {
  description = "Dataset Name"
  default     = "tf_demo_dataset"
}

variable "gcs_bucket_name" {
  description = "Bucket Name"
  default     = "bigquery-portfolio-tf-demo-bucket"
}

variable "gcs_storage_class" {
  description = "Bucket Storage Class"
  default     = "STANDARD"
}

and here's the revised main.tf using variables

terraform {
  required_providers {
    google = {
      source  = "hashicorp/google"
      version = "6.26.0"
    }
  }
}

provider "google" {
  # Configuration options
  project = var.project
  region  = var.region
  credentials = file(var.credentials)
}

resource "google_storage_bucket" "tf-demo-bucket" {
  name                     = var.gcs_bucket_name
  location                 = var.location
  storage_class            = var.gcs_storage_class
  force_destroy            = true
  public_access_prevention = "enforced"

  lifecycle_rule {
    condition {
      age = 1
    }
    action {
      type = "AbortIncompleteMultipartUpload"
    }
  }
}

resource "google_bigquery_dataset" "tf_demo_dataset" {
  dataset_id = var.bq_dataset_name
  location   = var.location
}
